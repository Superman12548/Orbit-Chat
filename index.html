-- Orbit Chat: database setup -- Run this entire file in Supabase Dashboard -> SQL Editor -> New query. -- This sets up the tables, access rules, group invites, and admin tools. -- It does not replace index.html; that file is the website. create extension if not exists pgcrypto; create table if not exists public.profiles ( id uuid primary key references auth.users(id) on delete cascade, username text unique, display_name text, created_at timestamptz not null default now() ); create table if not exists public.chats ( id uuid primary key default gen_random_uuid(), name text, is_group boolean not null default false, created_by uuid not null references auth.users(id) on delete cascade, created_at timestamptz not null default now() ); create table if not exists public.chat_members ( chat_id uuid not null references public.chats(id) on delete cascade, user_id uuid not null references auth.users(id) on delete cascade, created_at timestamptz not null default now(), primary key (chat_id, user_id) ); create table if not exists public.messages ( id uuid primary key default gen_random_uuid(), chat_id uuid not null references public.chats(id) on delete cascade, sender_id uuid not null references auth.users(id) on delete cascade, content text not null check (char_length(content) between 1 and 5000), created_at timestamptz not null default now() ); create table if not exists public.orbit_user_roles ( user_id uuid primary key references auth.users(id) on delete cascade, role text not null default 'member' check (role in ('member','admin')), updated_at timestamptz not null default now() ); alter table public.profiles enable row level security; alter table public.chats enable row level security; alter table public.chat_members enable row level security; alter table public.messages enable row level security; alter table public.orbit_user_roles enable row level security; -- Security-definer helpers avoid recursive RLS checks on membership tables. create or replace function public.orbit_is_admin() returns boolean language sql stable security definer set search_path = '' as $$ select exists (select 1 from public.orbit_user_roles r where r.user_id = auth.uid() and r.role = 'admin'); $$; revoke all on function public.orbit_is_admin() from public; grant execute on function public.orbit_is_admin() to authenticated; create or replace function public.orbit_can_access_chat(target_chat_id uuid) returns boolean language sql stable security definer set search_path = '' as $$ select exists ( select 1 from public.chats c where c.id = target_chat_id and ( exists (select 1 from public.chat_members cm where cm.chat_id = c.id and cm.user_id = auth.uid()) or (c.is_group = true and public.orbit_is_admin()) ) ); $$; revoke all on function public.orbit_can_access_chat(uuid) from public; grant execute on function public.orbit_can_access_chat(uuid) to authenticated; create or replace function public.orbit_can_view_role(target_user_id uuid) returns boolean language sql stable security definer set search_path = '' as $$ select target_user_id = auth.uid() or public.orbit_is_admin() or exists ( select 1 from public.chat_members mine join public.chat_members theirs on theirs.chat_id = mine.chat_id join public.chats c on c.id = mine.chat_id where mine.user_id = auth.uid() and theirs.user_id = target_user_id and c.is_group = true ); $$; revoke all on function public.orbit_can_view_role(uuid) from public; grant execute on function public.orbit_can_view_role(uuid) to authenticated; -- Remove/recreate only this setup's named policies. Other policies are not dropped. drop policy if exists "orbit profiles read directory" on public.profiles; create policy "orbit profiles read directory" on public.profiles for select to authenticated using (true); drop policy if exists "orbit profiles insert self" on public.profiles; create policy "orbit profiles insert self" on public.profiles for insert to authenticated with check (id = auth.uid()); drop policy if exists "orbit profiles update self" on public.profiles; create policy "orbit profiles update self" on public.profiles for update to authenticated using (id = auth.uid()) with check (id = auth.uid()); -- Authenticated users can create chats as themselves and read chats they belong to. drop policy if exists "orbit chats create own" on public.chats; create policy "orbit chats create own" on public.chats for insert to authenticated with check (created_by = auth.uid()); drop policy if exists "orbit chats read accessible" on public.chats; create policy "orbit chats read accessible" on public.chats for select to authenticated using (created_by = auth.uid() or public.orbit_can_access_chat(id)); drop policy if exists "orbit admins read all groups" on public.chats; create policy "orbit admins read all groups" on public.chats for select to authenticated using (is_group = true and public.orbit_is_admin()); -- A creator can add members to a chat they created (needed for DMs and group creator). drop policy if exists "orbit chat creator adds members" on public.chat_members; create policy "orbit chat creator adds members" on public.chat_members for insert to authenticated with check ( exists (select 1 from public.chats c where c.id = chat_id and c.created_by = auth.uid()) ); drop policy if exists "orbit members read accessible memberships" on public.chat_members; create policy "orbit members read accessible memberships" on public.chat_members for select to authenticated using ( user_id = auth.uid() or public.orbit_can_access_chat(chat_id) ); -- Only participants can read messages; admins may read group messages without joining. drop policy if exists "orbit participants read messages" on public.messages; create policy "orbit participants read messages" on public.messages for select to authenticated using (public.orbit_can_access_chat(chat_id)); drop policy if exists "orbit participants send messages" on public.messages; create policy "orbit participants send messages" on public.messages for insert to authenticated with check ( sender_id = auth.uid() and exists ( select 1 from public.chat_members cm where cm.chat_id = messages.chat_id and cm.user_id = auth.uid() ) ); -- Users see their own role, admins see all roles, and group members see fellow members' roles. drop policy if exists "orbit roles visible to authorized users" on public.orbit_user_roles; create policy "orbit roles visible to authorized users" on public.orbit_user_roles for select to authenticated using (public.orbit_can_view_role(user_id)); drop policy if exists "orbit roles no client writes" on public.orbit_user_roles; create policy "orbit roles no client writes" on public.orbit_user_roles for all to authenticated using (false) with check (false); create or replace function public.orbit_invite_to_group(target_chat_id uuid, target_user_id uuid) returns void language plpgsql security definer set search_path = '' as $$ declare creator_id uuid; is_group_chat boolean; begin if auth.uid() is null then raise exception 'Sign in before inviting people'; end if; select c.created_by, c.is_group into creator_id, is_group_chat from public.chats c where c.id = target_chat_id; if not found or is_group_chat is distinct from true then raise exception 'Group not found'; end if; if creator_id <> auth.uid() then raise exception 'Only the group creator can invite people'; end if; if not exists (select 1 from auth.users u where u.id = target_user_id) then raise exception 'That account does not exist'; end if; insert into public.chat_members(chat_id, user_id) values (target_chat_id, target_user_id) on conflict do nothing; end; $$; revoke all on function public.orbit_invite_to_group(uuid, uuid) from public; grant execute on function public.orbit_invite_to_group(uuid, uuid) to authenticated; create or replace function public.orbit_set_user_role(target_user_id uuid, new_role text) returns void language plpgsql security definer set search_path = '' as $$ declare admin_count integer; begin if auth.uid() is null or not public.orbit_is_admin() then raise exception 'Only an admin can change member roles'; end if; if new_role not in ('admin','member') then raise exception 'Role must be admin or member'; end if; if target_user_id = auth.uid() and new_role = 'member' then select count(*) into admin_count from public.orbit_user_roles where role = 'admin'; if admin_count <= 1 then raise exception 'You cannot remove the last admin'; end if; end if; insert into public.orbit_user_roles(user_id, role, updated_at) values (target_user_id, new_role, now()) on conflict (user_id) do update set role = excluded.role, updated_at = now(); end; $$; revoke all on function public.orbit_set_user_role(uuid, text) from public; grant execute on function public.orbit_set_user_role(uuid, text) to authenticated; create or replace function public.orbit_delete_chat(target_chat_id uuid) returns void language plpgsql security definer set search_path = '' as $$ declare target_is_group boolean; begin if auth.uid() is null or not public.orbit_is_admin() then raise exception 'Only an admin can delete conversations'; end if; select c.is_group into target_is_group from public.chats c where c.id = target_chat_id; if not found then raise exception 'Conversation not found'; end if; if target_is_group is distinct from true then raise exception 'Admins can only delete group chats with this tool'; end if; delete from public.messages where chat_id = target_chat_id; delete from public.chat_members where chat_id = target_chat_id; delete from public.chats where id = target_chat_id; end; $$; revoke all on function public.orbit_delete_chat(uuid) from public; grant execute on function public.orbit_delete_chat(uuid) to authenticated;
