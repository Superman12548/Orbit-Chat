-- Orbit Chat role security and admin tools -- Run this in Supabase Dashboard -> SQL Editor -> New query. -- Make sure the account ambrosewang@westminster.net already exists in Auth before running this. create table if not exists public.orbit_user_roles ( user_id uuid primary key references auth.users(id) on delete cascade, role text not null default 'member' check (role in ('member', 'admin')), updated_at timestamptz not null default now() ); alter table public.orbit_user_roles enable row level security; create or replace function public.orbit_is_admin() returns boolean language sql stable security definer set search_path = public as $$ select exists ( select 1 from public.orbit_user_roles r where r.user_id = auth.uid() and r.role = 'admin' ); $$; revoke all on function public.orbit_is_admin() from public; grant execute on function public.orbit_is_admin() to authenticated; drop policy if exists "orbit roles read self or admin" on public.orbit_user_roles; create policy "orbit roles read self or admin" on public.orbit_user_roles for select to authenticated using (user_id = auth.uid() or public.orbit_is_admin()); -- The frontend cannot directly insert/update/delete roles. Role changes must use the RPC below. drop policy if exists "orbit roles no client writes" on public.orbit_user_roles; create policy "orbit roles no client writes" on public.orbit_user_roles for all to authenticated using (false) with check (false); -- Let admins see the member directory used by the admin panel. alter table public.profiles enable row level security; drop policy if exists "orbit admins can view profiles" on public.profiles; create policy "orbit admins can view profiles" on public.profiles for select to authenticated using (public.orbit_is_admin()); create or replace function public.orbit_set_user_role(target_user_id uuid, new_role text) returns void language plpgsql security definer set search_path = public as $$ declare admin_count integer; begin if auth.uid() is null or not public.orbit_is_admin() then raise exception 'Only an admin can change member roles'; end if; if new_role not in ('admin', 'member') then raise exception 'Role must be admin or member'; end if; if target_user_id = auth.uid() and new_role = 'member' then select count(*) into admin_count from public.orbit_user_roles where role = 'admin'; if admin_count <= 1 then raise exception 'You cannot remove the last admin'; end if; end if; insert into public.orbit_user_roles(user_id, role, updated_at) values (target_user_id, new_role, now()) on conflict (user_id) do update set role = excluded.role, updated_at = now(); end; $$; revoke all on function public.orbit_set_user_role(uuid, text) from public; grant execute on function public.orbit_set_user_role(uuid, text) to authenticated; create or replace function public.orbit_delete_chat(target_chat_id uuid) returns void language plpgsql security definer set search_path = public as $$ begin if auth.uid() is null or not public.orbit_is_admin() then raise exception 'Only an admin can delete conversations'; end if; delete from public.messages where chat_id = target_chat_id; delete from public.chat_members where chat_id = target_chat_id; delete from public.chats where id = target_chat_id; end; $$; revoke all on function public.orbit_delete_chat(uuid) from public;
